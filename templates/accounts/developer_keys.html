Verifying Webhook Signatures (Security Best Practice)

TextBridge signs every outgoing webhook payload using HMAC-SHA256 with your secret key in the X-TextBridge-Signature header. Verify it on your server to prevent spoofing. 

import hmac
import hashlib

WEBHOOK_SECRET = "whsec_YOUR_SECRET_KEY"

def verify_textbridge_webhook(raw_request_body: bytes, signature_header: str) -> bool:
    expected_sig = hmac.new(
        WEBHOOK_SECRET.encode('utf-8'),
        raw_request_body,
        hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected_sig, signature_header)
const crypto = require('crypto');

const WEBHOOK_SECRET = 'whsec_YOUR_SECRET_KEY';

function verifyTextBridgeWebhook(rawRequestBody, signatureHeader) {
  const expectedSig = crypto
    .createHmac('sha256', WEBHOOK_SECRET)
    .update(rawRequestBody)
    .digest('hex');
  return crypto.timingSafeEqual(Buffer.from(expectedSig), Buffer.from(signatureHeader));
}

import requests

API_KEY = "tb_live_YOUR_SECRET_KEY"
url = "http://127.0.0.1:8000/api/v1/sms/send"

headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json"
}

payload = {
    "recipient": "0244123456",
    "message": "Your verification code is 849204",
    "sender_id": "Abtuniverse"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
const apiKey = 'tb_live_YOUR_SECRET_KEY';

fetch('http://127.0.0.1:8000/api/v1/sms/send', {
  method: 'POST',
  headers: {
    'Authorization': `Bearer ${apiKey}`,
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    recipient: '0244123456',
    message: 'Your verification code is 849204',
    sender_id: 'Abtuniverse'
  })
})
.then(res => res.json())
.then(data => console.log(data))
.catch(err => console.error(err));
<?php
$apiKey = "tb_live_YOUR_SECRET_KEY";
$data = [
    "recipient" => "0244123456",
    "message"   => "Your verification code is 849204",
    "sender_id" => "Abtuniverse"
];

$ch = curl_init("http://127.0.0.1:8000/api/v1/sms/send");

    "Authorization: Bearer " . $apiKey,
    "Content-Type: application/json"
]);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));

$response = curl_exec($ch);
curl_close($ch);
