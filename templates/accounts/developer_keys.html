import requests

API_KEY = "tb_live_YOUR_SECRET_KEY"
url = "http://127.0.0.1:8000/api/v1/sms/send"

headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json"
}

payload = {
    "recipient": "0244123456",
    "message": "Your verification code is 849204",
    "sender_id": "Abtuniverse"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
const apiKey = 'tb_live_YOUR_SECRET_KEY';

fetch('http://127.0.0.1:8000/api/v1/sms/send', {
  method: 'POST',
  headers: {
    'Authorization': `Bearer ${apiKey}`,
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    recipient: '0244123456',
    message: 'Your verification code is 849204',
    sender_id: 'Abtuniverse'
  })
})
.then(res => res.json())
.then(data => console.log(data))
.catch(err => console.error(err));
<?php
$apiKey = "tb_live_YOUR_SECRET_KEY";
$data = [
    "recipient" => "0244123456",
    "message"   => "Your verification code is 849204",
    "sender_id" => "Abtuniverse"
];

$ch = curl_init("http://127.0.0.1:8000/api/v1/sms/send");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    "Authorization: Bearer " . $apiKey,
    "Content-Type: application/json"
]);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));

$response = curl_exec($ch);
curl_close($ch);
echo $response;
